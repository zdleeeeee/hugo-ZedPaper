{{- /* Normalize legacy names and author objects into a shared list. */ -}} {{- $authors := slice -}} {{- with . -}} {{- $entries := slice . -}} {{- if reflect.IsSlice . -}} {{- $entries = . -}} {{- end -}} {{- range $entries -}} {{- $name := "" -}} {{- $url := "" -}} {{- $email := "" -}} {{- if reflect.IsMap . -}} {{- $name = .name | default "" -}} {{- $url = .url | default "" -}} {{- $email = .email | default "" -}} {{- else -}} {{- $name = . -}} {{- end -}} {{- /* Author paths belong to this site, including when hosted in a subdirectory. */ -}} {{- if and (strings.HasPrefix $url "/") (not (strings.HasPrefix $url "//")) -}} {{- $basePath := (urls.Parse site.BaseURL).Path | default "/" -}} {{- if not (strings.HasPrefix $url $basePath) -}} {{- $url = strings.TrimPrefix "/" $url | relURL -}} {{- end -}} {{- end -}} {{- with strings.TrimSpace $name -}} {{- $authors = $authors | append (dict "name" . "url" $url "email" $email) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $authors -}}
