{{- /* Normalize legacy names and author objects into a shared list. */ -}} {{- $authors := slice -}} {{- with . -}} {{- $entries := slice . -}} {{- if reflect.IsSlice . -}} {{- $entries = . -}} {{- end -}} {{- range $entries -}} {{- $name := "" -}} {{- $url := "" -}} {{- $email := "" -}} {{- if reflect.IsMap . -}} {{- $name = .name | default "" -}} {{- $url = .url | default "" -}} {{- $email = .email | default "" -}} {{- else -}} {{- $name = . -}} {{- end -}} {{- with strings.TrimSpace $name -}} {{- $authors = $authors | append (dict "name" . "url" $url "email" $email) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $authors -}}
